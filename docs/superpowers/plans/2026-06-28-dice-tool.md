**Architecture:** Hilt + ViewModel + Repository (UDF). Hilt provides `DiceRepository` (wrapping DataStore) and `DiceViewModel`. Compose screens collect `StateFlow<DiceUiState>`. Tool navbar (Tools | Options) replaces FAB on all tool screens.

**Tech Stack:** Kotlin, Jetpack Compose, Material 3, Hilt, DataStore Preferences, kotlinx.serialization, Navigation Compose

```kotlin
package com.turnzero.tools.dice

import androidx.datastore.core.DataStore
import androidx.datastore.preferences.core.Preferences
import androidx.datastore.preferences.core.edit
import com.turnzero.core.data.UserPreferences
import kotlinx.coroutines.flow.first
import kotlinx.coroutines.flow.map
import kotlinx.serialization.encodeToString
import kotlinx.serialization.json.Json
import javax.inject.Inject
import javax.inject.Singleton

@Singleton
class DataStoreDiceRepo @Inject constructor(
    private val dataStore: DataStore<Preferences>
) : DiceRepo {

    private val json = Json { ignoreUnknownKeys = true }

    override suspend fun getPool(gameId: String): List<Die> {
        val key = UserPreferences.dicePoolKey(gameId)
        val raw = dataStore.data.map { it[key] }.first() ?: return emptyList()
        return try {
            json.decodeFromString<List<Die>>(raw)
        } catch (_: Exception) {
            emptyList()
        }
    }

    override suspend fun savePool(gameId: String, dice: List<Die>) {
        val key = UserPreferences.dicePoolKey(gameId)
        dataStore.edit { prefs ->
            prefs[key] = json.encodeToString(dice)
        }
    }

    override suspend fun getSelectedColor(): DiceColor {
        val raw = dataStore.data.map { it[UserPreferences.DICE_SELECTED_COLOR] }.first()
        return raw?.let { name ->
            try { DiceColor.valueOf(name) } catch (_: Exception) { DiceColor.RED }
        } ?: DiceColor.RED
    }

    override suspend fun saveSelectedColor(color: DiceColor) {
        dataStore.edit { prefs ->
            prefs[UserPreferences.DICE_SELECTED_COLOR] = color.name
        }
    }

    override suspend fun getHistory(): List<DiceRoll> {
        val raw = dataStore.data.map { it[UserPreferences.DICE_HISTORY] }.first() ?: return emptyList()
        return try {
            json.decodeFromString<List<DiceRoll>>(raw)
        } catch (_: Exception) {
            emptyList()
        }
    }

    override suspend fun saveHistory(history: List<DiceRoll>) {
        val capped = history.take(50)
        dataStore.edit { prefs ->
            prefs[UserPreferences.DICE_HISTORY] = json.encodeToString(capped)
        }
    }
}
```

```kotlin
package com.turnzero.tools.dice

import androidx.lifecycle.SavedStateHandle
import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import dagger.hilt.android.lifecycle.HiltViewModel
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch
import java.util.UUID
import javax.inject.Inject
import kotlin.random.Random

@HiltViewModel
class DiceViewModel @Inject constructor(
    private val diceRepo: DiceRepo,
    private val savedStateHandle: SavedStateHandle
) : ViewModel() {

    private val gameId: String = savedStateHandle.get<String>("gameId") ?: "*"

    private val _uiState = MutableStateFlow(DiceUiState(gameId = gameId))
    val uiState: StateFlow<DiceUiState> = _uiState.asStateFlow()

    init {
        viewModelScope.launch {
            val pool = diceRepo.getPool(gameId)
            val color = diceRepo.getSelectedColor()
            val history = diceRepo.getHistory()
            val presets = getPresetsForGame(gameId)
            _uiState.update {
                it.copy(
                    dice = pool,
                    selectedColor = color,
                    history = history,
                    presets = presets
                )
            }
        }
    }

    fun changeDieCount(faceCount: Int, color: DiceColor, delta: Int) {
        val current = _uiState.value.dice
        val matching = current.filter { it.faceCount == faceCount && it.color == color }
        if (delta > 0) {
            val newDice = (1..delta).map {
                Die(id = UUID.randomUUID().toString(), faceCount = faceCount, color = color)
            }
            val updated = current + newDice
            persistPool(updated)
        } else {
            val removeCount = minOf(-delta, matching.size)
            val toRemove = matching.take(removeCount).map { it.id }.toSet()
            val updated = current.filter { it.id !in toRemove }
            persistPool(updated)
        }
    }

    fun setSelectedColor(color: DiceColor) {
        _uiState.update { it.copy(selectedColor = color) }
        viewModelScope.launch { diceRepo.saveSelectedColor(color) }
    }

    fun roll() {
        val dice = _uiState.value.dice
        if (dice.isEmpty()) return
        val results = dice.map { Random.nextInt(1, it.faceCount + 1) }
        val total = results.sum()
        _uiState.update { it.copy(isRolling = true, results = results, total = total) }
    }

    fun animationComplete() {
        val state = _uiState.value
        val results = state.results ?: return
        val dice = state.dice
        val roll = DiceRoll(
            id = UUID.randomUUID().toString(),
            timestamp = System.currentTimeMillis(),
            dice = dice,
            results = results,
            total = results.sum()
        )
        val updatedHistory = listOf(roll) + state.history
        _uiState.update { it.copy(isRolling = false, history = updatedHistory) }
        viewModelScope.launch { diceRepo.saveHistory(updatedHistory) }
    }

    fun loadPreset(preset: DicePreset) {
        persistPool(preset.dice)
    }

    fun clearHistory() {
        _uiState.update { it.copy(history = emptyList()) }
        viewModelScope.launch { diceRepo.saveHistory(emptyList()) }
    }

    private fun persistPool(dice: List<Die>) {
        _uiState.update { it.copy(dice = dice) }
        viewModelScope.launch { diceRepo.savePool(gameId, dice) }
    }
}
```

```kotlin
import androidx.compose.animation.core.animateFloatAsState

import androidx.compose.ui.unit.dp
import androidx.compose.ui.unit.sp
import kotlinx.coroutines.delay
import kotlin.random.Random

@Composable
fun DiceAnimation(
    dice: List<Die>,
    results: List<Int>?,
    isRolling: Boolean,
    onAnimationComplete: () -> Unit,
    modifier: Modifier = Modifier
) {
    val hasRolled = results != null

    var displayedNumbers by remember { mutableStateOf(List(dice.size) { 1 }) }
    var localResults by remember { mutableStateOf(results) }

    LaunchedEffect(isRolling, results) {
        if (isRolling && results != null) {
            localResults = results
            // Fast shuffle phase: ~600ms
            val fastEnd = System.currentTimeMillis() + 600
            while (System.currentTimeMillis() < fastEnd) {
                displayedNumbers = dice.map { Random.nextInt(1, it.faceCount + 1) }
                delay(50)
            }
            // Deceleration phase: ~400ms
            delay(100)
            displayedNumbers = dice.mapIndexed { i, _ ->
                if (Random.nextBoolean()) results[i] else Random.nextInt(1, dice[i].faceCount + 1)
            }
            delay(200)
            displayedNumbers = dice.mapIndexed { i, _ ->
                if (Random.nextBoolean()) results[i] else Random.nextInt(1, dice[i].faceCount + 1)
            }
            delay(400)
            displayedNumbers = results
            onAnimationComplete()
        } else if (!isRolling && results != null && displayedNumbers != results) {
            displayedNumbers = results
        }
    }

    Row(
        modifier = modifier,
        horizontalArrangement = Arrangement.Center,
        verticalAlignment = Alignment.CenterVertically
    ) {
        dice.forEachIndexed { index, die ->
            val number = if (isRolling || hasRolled) displayedNumbers.getOrElse(index) { 1 } else 0
            val settled = !isRolling && results != null

            val scale by animateFloatAsState(
                targetValue = if (settled) 1f else 1f,
                animationSpec = spring(dampingRatio = 0.4f, stiffness = 300f),
                label = "dieScale"
            )

            DieCard(
                die = die,
                displayNumber = if (hasRolled || isRolling) number else null,
                modifier = Modifier.padding(horizontal = 6.dp)
            )
        }
    }
}

@Composable

    displayNumber: Int?,

) {

    val faceLabel = "d${die.faceCount}"

            color = dieColor,
            fontSize = 10.sp
        )
        Spacer(modifier = Modifier.height(2.dp))
        Text(
            text = if (displayNumber != null) displayNumber.toString() else faceLabel,
            style = MaterialTheme.typography.headlineMedium,
            fontWeight = FontWeight.Bold,
            color = if (displayNumber != null) Color.Unspecified else dieColor.copy(alpha = 0.5f),
            textAlign = TextAlign.Center

    }
```
