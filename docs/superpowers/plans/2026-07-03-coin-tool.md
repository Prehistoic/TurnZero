package com.turnzero.tools.coin

import androidx.datastore.core.DataStore
import androidx.datastore.preferences.core.Preferences
import androidx.datastore.preferences.core.edit
import com.turnzero.core.data.UserPreferences
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.flow.map
import kotlinx.serialization.encodeToString
import kotlinx.serialization.json.Json
import javax.inject.Inject
import javax.inject.Singleton

@Singleton
class DataStoreCoinRepo @Inject constructor(
    private val dataStore: DataStore<Preferences>
) : CoinRepo {

    private val json = Json { ignoreUnknownKeys = true }

    override fun getCoinCount(gameId: String): Flow<Int> {
        val key = UserPreferences.coinCountKey(gameId)
        return dataStore.data.map { prefs ->
            prefs[key]?.toIntOrNull() ?: 1
        }
    }

    override suspend fun saveCoinCount(gameId: String, count: Int) {
        val key = UserPreferences.coinCountKey(gameId)
        dataStore.edit { prefs ->
            prefs[key] = count.toString()
        }
    }

    override fun getHistory(): Flow<List<CoinFlip>> {
        return dataStore.data.map { prefs ->
            val raw = prefs[UserPreferences.COIN_HISTORY] ?: return@map emptyList()
            try {
                json.decodeFromString<List<CoinFlip>>(raw)
            } catch (_: Exception) {
                emptyList()
            }
        }
    }

    override suspend fun saveHistory(history: List<CoinFlip>) {
        val capped = history.take(10)
        dataStore.edit { prefs ->
            prefs[UserPreferences.COIN_HISTORY] = json.encodeToString(capped)
        }
    }
}

package com.turnzero.tools.coin

import androidx.lifecycle.SavedStateHandle
import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import dagger.hilt.android.lifecycle.HiltViewModel
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch
import java.util.UUID
import javax.inject.Inject
import kotlin.random.Random

@HiltViewModel
class CoinViewModel @Inject constructor(
    private val coinRepo: CoinRepo,
    private val savedStateHandle: SavedStateHandle
) : ViewModel() {

    private val gameId: String = savedStateHandle.get<String>("gameId") ?: "*"

    private val _uiState = MutableStateFlow(CoinUiState(gameId = gameId))
    val uiState: StateFlow<CoinUiState> = _uiState.asStateFlow()

    init {
        viewModelScope.launch {
            coinRepo.getCoinCount(gameId).collect { count ->
                _uiState.update { it.copy(coinCount = count.coerceIn(1, 8)) }
            }
        }
        viewModelScope.launch {
            val presets = getCoinPresets(gameId)
            _uiState.update { it.copy(presets = presets) }
        }
        viewModelScope.launch {
            coinRepo.getHistory().collect { history ->
                _uiState.update { it.copy(history = history) }
            }
        }
    }

    fun setCoinCount(count: Int) {
        val clamped = count.coerceIn(1, 8)
        _uiState.update { it.copy(coinCount = clamped) }
        viewModelScope.launch { coinRepo.saveCoinCount(gameId, clamped) }
    }

    fun flip() {
        val coinCount = _uiState.value.coinCount
        val results = (1..coinCount).map {
            if (Random.nextBoolean()) CoinFace.HEAD else CoinFace.TAIL
        }
        _uiState.update {
            it.copy(
                isFlipping = true,
                results = results,
                headsCount = results.count { face -> face == CoinFace.HEAD },
                tailsCount = results.count { face -> face == CoinFace.TAIL }
            )
        }
    }

    fun flipAnimationComplete() {
        val state = _uiState.value
        val flip = CoinFlip(
            id = UUID.randomUUID().toString(),
            timestamp = System.currentTimeMillis(),
            coinCount = state.coinCount,
            results = state.results,
            headsCount = state.headsCount,
            tailsCount = state.tailsCount
        )
        val updatedHistory = (listOf(flip) + state.history).take(10)
        _uiState.update { it.copy(isFlipping = false, history = updatedHistory) }
        viewModelScope.launch { coinRepo.saveHistory(updatedHistory) }
    }

    fun loadPreset(preset: CoinPreset) {
        setCoinCount(preset.coinCount)
    }

    fun clearHistory() {
        _uiState.update { it.copy(history = emptyList()) }
        viewModelScope.launch { coinRepo.saveHistory(emptyList()) }
    }

    fun toggleHistory() {
        _uiState.update { it.copy(showHistory = !it.showHistory) }
    }
}

import androidx.compose.ui.draw.scale
import androidx.compose.ui.graphics.Brush
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.graphics.graphicsLayer
import androidx.compose.ui.platform.LocalDensity

import androidx.compose.ui.unit.dp
import androidx.compose.ui.unit.sp
import kotlinx.coroutines.delay
import kotlin.random.Random

@Composable
fun CoinAnimation(
    coinCount: Int,
    results: List<CoinFace>,
    isFlipping: Boolean,
    onAnimationComplete: () -> Unit,
    modifier: Modifier = Modifier,
    coinSize: Dp,
    onFlip: () -> Unit = {},
    triggerScale: Int = 0
) {
    val hasFlipped = results.isNotEmpty()

    var displayedFaces by remember { mutableStateOf(List(coinCount) { CoinFace.HEAD }) }

    LaunchedEffect(isFlipping, results) {
        if (isFlipping && results.isNotEmpty()) {
            val fastEnd = System.currentTimeMillis() + 600
            while (System.currentTimeMillis() < fastEnd) {
                displayedFaces = List(coinCount) { if (Random.nextBoolean()) CoinFace.HEAD else CoinFace.TAIL }
                delay(80)
            }
            delay(100)
            displayedFaces = List(coinCount) { if (Random.nextBoolean()) CoinFace.HEAD else CoinFace.TAIL }
            delay(150)
            displayedFaces = results.mapIndexed { i, _ ->
                if (Random.nextBoolean()) results[i] else (if (Random.nextBoolean()) CoinFace.HEAD else CoinFace.TAIL)
            }
            delay(250)
            displayedFaces = results
            onAnimationComplete()
        } else if (!isFlipping && results.isNotEmpty() && displayedFaces != results) {
            displayedFaces = results
        }
    }

    var targetScale by remember { mutableFloatStateOf(1f) }

    LaunchedEffect(triggerScale) {
        if (triggerScale > 0) {
            targetScale = 1.08f
            delay(100)
            targetScale = 1f
            delay(80)
            onFlip()
        }
    }

    val animatedScale by animateFloatAsState(
        targetValue = targetScale,
        animationSpec = spring(dampingRatio = 0.5f, stiffness = 500f),
        label = "tapScale"
    )

    val density = LocalDensity.current

    val rows = coinCount.chunked(4)

    Column(
        modifier = modifier.scale(animatedScale),
        horizontalAlignment = Alignment.CenterHorizontally,
        verticalArrangement = Arrangement.Center
    ) {
        rows.forEachIndexed { rowIndex, rowCount ->
            Row(
                horizontalArrangement = Arrangement.Center,
                verticalAlignment = Alignment.CenterVertically
            ) {
                repeat(rowCount) { colIndex ->
                    val index = rowIndex * 4 + colIndex
                    val face = if (isFlipping || hasFlipped) displayedFaces.getOrElse(index) { CoinFace.HEAD } else CoinFace.HEAD

                    CoinCard(
                        face = face,
                        isAnimating = isFlipping,
                        modifier = Modifier.padding(4.dp),
                        coinSize = coinSize
                    )
                }
            }
        }
    }
}

@Composable
fun CoinCard(
    face: CoinFace,
    isAnimating: Boolean,
    modifier: Modifier = Modifier,
    coinSize: Dp
) {
    val isHead = face == CoinFace.HEAD

    val headGradient = Brush.radialGradient(
        colors = listOf(Color(0xFFFFD700), Color(0xFFDAA520), Color(0xFFB8860B))
    )
    val tailGradient = Brush.radialGradient(
        colors = listOf(Color(0xFFE0E0E0), Color(0xFFC0C0C0), Color(0xFFA0A0A0))
    )

    val gradient = if (isHead) headGradient else tailGradient
    val accentColor = if (isHead) Color(0xFF8B6914) else Color(0xFF606060)
    val emoji = if (isHead) "\uD83D\uDC64" else "⭐"

    val rimColor = if (isHead) Color(0xFFD4AF37) else Color(0xFFB0B0B0)

    var targetRotX by remember { mutableFloatStateOf(0f) }
    var targetRotY by remember { mutableFloatStateOf(0f) }

    LaunchedEffect(isAnimating) {
        if (isAnimating) {
            targetRotX = Random.nextFloat() * 720f
            targetRotY = Random.nextFloat() * 720f
        } else {
            targetRotX = 0f
            targetRotY = 0f
        }
    }

    val density = LocalDensity.current
    val cameraDist = 8f * density.density

    val rotX by animateFloatAsState(
        targetValue = targetRotX,
        animationSpec = tween(durationMillis = 800),
        label = "rotX"
    )
    val rotY by animateFloatAsState(
        targetValue = targetRotY,
        animationSpec = tween(durationMillis = 1000),
        label = "rotY"
    )

    val borderWidth = (coinSize.value / 32f).dp

    Box(
        modifier = modifier
            .size(coinSize)
            .graphicsLayer {
                rotationX = rotX
                rotationY = rotY
                cameraDistance = cameraDist
            }
            .border(borderWidth, rimColor, CircleShape)
            .background(gradient, CircleShape),
        contentAlignment = Alignment.Center
    ) {
        Text(
            text = emoji,
            fontSize = (coinSize.value * 0.38f).sp,
            textAlign = TextAlign.Center
        )
    }

package com.turnzero.tools.coin

import androidx.compose.foundation.clickable
import androidx.compose.foundation.interaction.MutableInteractionSource
import androidx.compose.foundation.layout.*
import androidx.compose.material3.*

import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.text.font.FontWeight
import androidx.compose.ui.unit.dp

@Composable
fun CoinScreen(
    uiState: CoinUiState,
    onSetCoinCount: (Int) -> Unit,
    onFlip: () -> Unit,
    onFlipAnimationComplete: () -> Unit,
    onLoadPreset: (CoinPreset) -> Unit,
    onClearHistory: () -> Unit,
    onToggleHistory: () -> Unit,
    modifier: Modifier = Modifier
) {
    val canFlip = !uiState.isFlipping

    var tapCount by remember { mutableIntStateOf(0) }

    Column(modifier = modifier.fillMaxSize()) {
        BoxWithConstraints(
            modifier = Modifier
                .weight(1f)
                .fillMaxWidth()
                .clickable(
                    indication = null,
                    interactionSource = remember { MutableInteractionSource() },
                    enabled = canFlip
                ) { tapCount++ },
            contentAlignment = Alignment.Center
        ) {
            val columns = uiState.coinCount.coerceAtMost(4)
            val widthFraction = when (columns) {
                1 -> 0.35f
                2 -> 0.60f
                else -> 0.8f
            }
            val availableWidth = maxWidth * widthFraction
            val gaps = 8.dp * (columns - 1)
            val coinSize = (availableWidth - gaps) / columns

            CoinAnimation(
                coinCount = uiState.coinCount,
                results = uiState.results,
                isFlipping = uiState.isFlipping,
                onAnimationComplete = onFlipAnimationComplete,
                onFlip = onFlip,
                coinSize = coinSize,
                triggerScale = tapCount
            )
        }

        Box(
            modifier = Modifier
                .fillMaxWidth()
                .weight(0.25f),
            contentAlignment = Alignment.Center
        ) {
            if (uiState.results.isNotEmpty() && !uiState.isFlipping) {
                Text(
                    text = "${uiState.headsCount} Heads · ${uiState.tailsCount} Tails",
                    style = MaterialTheme.typography.headlineMedium,
                    fontWeight = FontWeight.Bold
                )
            }
        }
    }

    if (uiState.showHistory) {
        CoinHistorySheet(
            history = uiState.history,
            onClear = onClearHistory,
            onDismiss = onToggleHistory
        )
    }
}

@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun CoinHistorySheet(
    history: List<CoinFlip>,
    onClear: () -> Unit,
    onDismiss: () -> Unit
) {
    ModalBottomSheet(onDismissRequest = onDismiss) {
        Column(modifier = Modifier.padding(horizontal = 16.dp)) {
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = Alignment.CenterVertically
            ) {
                Text(
                    text = "History",
                    style = MaterialTheme.typography.titleLarge
                )
                TextButton(onClick = onClear) {
                    Icon(Icons.Default.Delete, contentDescription = null, modifier = Modifier.size(18.dp))
                    Spacer(Modifier.width(4.dp))
                    Text("Clear")
                }
            }

            Spacer(Modifier.height(8.dp))

            if (history.isEmpty()) {
                Text(
                    text = "No flips yet",
                    style = MaterialTheme.typography.bodyMedium,
                    color = MaterialTheme.colorScheme.onSurfaceVariant
                )
            } else {
                LazyColumn {
                    itemsIndexed(history) { index, flip ->
                        FlipEntry(flip = flip, index = history.size - index)
                        if (index < history.lastIndex) {
                            HorizontalDivider(modifier = Modifier.padding(vertical = 4.dp))
                        }
                    }
                }
            }

            Spacer(Modifier.height(32.dp))
        }
    }
}

@Composable
private fun FlipEntry(flip: CoinFlip, index: Int) {
    val timeFormat = SimpleDateFormat("HH:mm", Locale.getDefault())
    val timeStr = timeFormat.format(Date(flip.timestamp))
    val summary = "${flip.coinCount} coin${if (flip.coinCount > 1) "s" else ""}"

    Row(
        modifier = Modifier
            .fillMaxWidth()
            .padding(vertical = 8.dp),
        horizontalArrangement = Arrangement.SpaceBetween,
        verticalAlignment = Alignment.CenterVertically
    ) {
        Text(
            text = "#$index  $summary",
            style = MaterialTheme.typography.labelMedium,
            fontWeight = FontWeight.Bold,
            modifier = Modifier.weight(1f)
        )
        Text(
            text = "${flip.headsCount}H · ${flip.tailsCount}T",
            style = MaterialTheme.typography.titleMedium,
            fontWeight = FontWeight.Bold,
            color = MaterialTheme.colorScheme.primary
        )
        Spacer(Modifier.width(12.dp))
        Text(
            text = timeStr,
            style = MaterialTheme.typography.labelSmall,
            color = MaterialTheme.colorScheme.onSurfaceVariant
        )
    }
}
